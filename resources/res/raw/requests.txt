 INSERT into units(name,code)
 select DISTINCT base_name,base_name  from requests_bills a, requests_bills_det b
 where a.status=-1 and a.tr_type in(1,2) and a.bill_sys=b.bill_id
 and not EXISTS(select * from units where name=(base_name))  ;

 INSERT into items(name,unit_id,item_type_id)
 select DISTINCT (item_name)  name,c.id unit_id,0  from requests_bills a, requests_bills_det b,units c
 where a.status=-1 and a.tr_type in(1,2) and a.bill_sys=b.bill_id
 and (b.item_unit )= c.name
 and not EXISTS(select * from items where name=(item_name))  ;
 update items set item_type_id=0 where item_type_id is null;

 update requests_bills_det
 set item_id=(select id from items where name=(item_name))
 ,item_type_id=(select item_type_id from items where name=(item_name))
 where item_id is null
 and EXISTS(select * from requests_bills where status=-1 and tr_type in(1,2) and bill_sys=requests_bills_det.bill_id) ;

 update requests_bills_det
 set base_unit=(select id from units where name=(base_name))
 ,unit_id=(select id from units where name=(unit_name))
 where EXISTS(select * from requests_bills where status=-1 and tr_type in(1,2) and bill_sys=requests_bills_det.bill_id) ;

  INSERT into unit_item( item_id,unit_id,u_val )
 select  DISTINCT item_id,unit_id,u_val from requests_bills_det a
 where EXISTS(select * from requests_bills where status=-1 and tr_type in(1,2) and bill_sys=a.bill_id)

 and not EXISTS(select * from unit_item b where b.item_id=a.item_id and a.unit_id=b.unit_id

 ) ;

 where  exists( select * from
 requests_bills a,bills b
  where a.status=-1 and a.tr_type in(1,2) and a.tr_type=b.tr_type
   and bill_transactions.bill_id=b.id
 and a.br_id=b.br_id and a.bill_no2=b.bill_no2
  and a.online_ref=b.bill_no2
 and strftime('%m',a.date_)=strftime('%m',b.date_)
 and a.cus_id=b.cus_id and b.online in(1,2) );



 delete from bills
 where online in(1,2)
 and exists( select * from
 requests_bills where status=-1 and tr_type in(1,2)
 and br_id=bills.br_id and bill_no2=bills.bill_no2
 and requests_bills.online_ref=bills.bill_no2

 insert into bills (id,tr_type,date_,amount,d_amount,cus_id,tran_status,bill_type,bill_no2,remarks,curr_id,br_id,
 cash_id,is_back,bill_no,curr_price,curr_mod,t_val,tax_amount,online,online_ref,time_,cost2,remarks,paid_amount)
 select bill_sys,tr_type,date_,amount,d_amount,cus_id,ifnull(tran_status,'2'),bill_type,bill_no2,remarks,curr_id,br_id,cash_id,is_back
 ,(case when a.is_back=1 then
 ifnull((select bill_sys from requests_bills where bill_id=a.bill_id and cus_id=a.cus_id and br_id=a.br_id),0)
 else  0 end) bill_no,a.curr_price,(case when a.curr_price=0 then 0 else 1 end ) curr_mod
 ,t_val,tax_amount,2,online_ref,time_,cost2,remarks,paid_amount
 from requests_bills a
 where a.status=-1 and a.tr_type in(1,2)
  and ROWID=(select max(rowid) from requests_bills b where a.bill_id=b.bill_id and a.tr_type=b.tr_type and a.status=b.status);
/*--Insert bill Details --*/
 insert into bill_transactions(bill_id,item_id,item_type_id,qty,qty_t,cost_price,sls_u_price,curr_id,d_amount,remark,unit_id,u_val,base_unit,qty_pr)
 select bill_id,item_id,ifnull(item_type_id,0),qty,qty_t,cost_price,sls_u_price,(select curr_id from requests_bills where requests_bills.bill_sys=a.bill_id) curr_id,d_amount2,remark2,
 unit_id,u_val,base_unit,qty_pr
 from requests_bills_det a
  where EXISTS(select * from requests_bills c where c.status=-1 and c.tr_type in(1,2) and c.bill_sys=a.bill_id
   and ROWID=(select max(rowid) from requests_bills b where c.bill_id=b.bill_id and c.tr_type=b.tr_type and c.status=b.status)
  )  ;

  /*--delete old item_price and insert new price only for sales--*/
   delete  from item_price where EXISTS
   (
   select *
   from requests_bills_det a,requests_bills c
  where  c.status=-1 and
  c.tr_type=1 and c.bill_sys=a.bill_id
  and sls_u_price!=0
  and item_price.item_id=a.item_id and item_price.unit_id=a.unit_id and c.curr_id=item_price.curr_id
   and a.sls_u_price*a.u_val!=item_price.sls_u_price
   );
   insert  or replace into item_price(item_id,curr_id,unit_id,SLS_U_PRICE)
   select distinct a.item_id,c.curr_id,a.unit_id,
   a.sls_u_price*a.u_val
    from requests_bills_det a,requests_bills c
   where  c.status=-1 and
   c.tr_type=1 and c.bill_sys=a.bill_id
   and sls_u_price!=0
    and not EXISTS(
     select * from item_price d
     where d.item_id=a.item_id and d.unit_id=a.unit_id and c.curr_id=d.curr_id and a.sls_u_price=d.sls_u_price);
